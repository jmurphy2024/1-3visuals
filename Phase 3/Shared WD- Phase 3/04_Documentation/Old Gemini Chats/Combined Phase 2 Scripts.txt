## Script: I-D_process_puma_gis.R
## Purpose: Determines the predominant PUMA for granular geographic areas (Tract, ZCTA, etc.)
##          by performing a spatial intersection using GIS shapefiles. This is a more robust
##          alternative to using tabular crosswalk files as it relies on true geographic
##          overlap rather than pre-computed tables, avoiding potential join key errors.
## Author: Your Name/Team Name
## Date Created: 2025-09-03
## Last Modified: 2025-09-03 (Added extensive documentation and step-by-step explanations)
## Dependencies: dplyr, here, sf, tigris, purrr
## Input: 01_data/processed/geographic_income_database_nhgis_summary.rds
## Output: 01_data/processed/geographic_income_database_nhgis_summary_with_puma.rds (UPDATED)

# ==== 0. SETUP & ENVIRONMENT PREPARATION ====

# ===== 0.1. Install and Load Required Packages =====
# This script relies on specialized packages for spatial data analysis. The following
# code checks if they are installed and, if not, installs them before loading.
#
# - `sf`: The modern standard for working with spatial "simple features" (vector data
#   like points, lines, and polygons) in R. It's used for all GIS operations.
# - `tigris`: A package that provides easy access to U.S. Census Bureau TIGER/Line
#   shapefiles, allowing us to download geographic boundaries directly into R.
# - `dplyr`, `purrr`, `here`: Core packages for data manipulation, iteration, and
#   robust file path management, consistent with the rest of the project.

if (!require(sf)) install.packages("sf")
if (!require(tigris)) install.packages("tigris")
if (!require(purrr)) install.packages("purrr")

library(dplyr)
library(here)
library(sf)
library(tigris)
library(purrr)

# ===== 0.2. Configure tigris Options =====
# Logic: Downloading large shapefiles can be slow. By setting `tigris_use_cache` to TRUE,
# the `tigris` package will save a local copy of any downloaded shapefile. On subsequent
# runs of this script, it will use the cached local version instead of re-downloading,
# which significantly speeds up the process.
options(tigris_use_cache = TRUE)


# ==== 1. LOAD NHGIS SUMMARY DATA ====
# Purpose: To load the master dataset that contains income and demographic information
# for all geographic areas. This is the file we will be adding the correct PUMA codes to.
#
# Logic: We use `here()` to construct a reliable file path relative to the project root.
# `stopifnot()` provides a clear, immediate error if the input file from the previous
# script (`I-A`) does not exist, preventing the script from running with missing data.

message("Loading NHGIS summary data...")
nhgis_summary_path <- here("01_data", "processed", "geographic_income_database_nhgis_summary.rds")
stopifnot("NHGIS summary data not found at the specified path. Please run script I-A first." = file.exists(nhgis_summary_path))
nhgis_data <- readRDS(nhgis_summary_path)
message("NHGIS summary data loaded successfully.")


# ==== 2. ACQUIRE AND PREPARE GEOGRAPHIC SHAPEFILES ====
# Purpose: To download the official boundary files (shapefiles) for all the geographic
# areas we need to analyze. This includes the PUMAs (our target geography) and all the
# smaller, "source" geographies (like tracts and school districts).
#
# Logic: We use the `tigris` package to directly and reliably download these files from
# the U.S. Census Bureau.

message("Downloading required shapefiles from the Census Bureau via tigris...")

# ===== 2.1. Define a Consistent Coordinate Reference System (CRS) =====
# What: A Coordinate Reference System (CRS) is a framework used to define how a
# two-dimensional, flat surface of a map relates to the real, three-dimensional Earth.
#
# Why it's critical:
#   1. Comparability: All spatial data must be in the same CRS to be compared or overlaid correctly.
#   2. Area Calculation: Latitude/longitude coordinates are angles, not uniform units of
#      distance. To accurately calculate area (which is essential for our analysis), we
#      must convert the data to a "projected" CRS that uses linear units like meters.
#
# Logic: We select the "US National Atlas Equal Area" projection (EPSG code: 2163).
# This CRS is specifically designed to preserve area measurements across the contiguous U.S.,
# making it ideal for our task of finding the largest overlapping area. `st_transform()`
# from the `sf` package will be used to convert all shapefiles to this CRS.
target_crs <- st_crs(2163)

# ===== 2.2. Download Shapefiles State-by-State =====
# Logic: For recent years, the `tigris` package requires downloading many geographies
# (like tracts and PUMAs) on a state-by-state basis. To get a national dataset,
# we first get a list of all state FIPS codes and then loop through them, downloading
# the data for each state and binding them together into a single national file.
# We exclude some territories that may not have all the required data types to avoid errors.

states_to_download <- states(year = 2020)$STATEFP
states_to_download <- states_to_download[!states_to_download %in% c("60", "66", "69", "78")] # Exclude Pacific territories

# This helper function simplifies the state-by-state download process.
download_national_file <- function(tigris_function, year) {
  message(paste("-> Downloading national file for:", deparse(substitute(tigris_function))))
  # `map_dfr` from purrr iterates through each state FIPS code, calls the specified
  # tigris function (e.g., `pumas`), and binds the results into a single `sf` data frame.
  map_dfr(states_to_download, ~ tigris_function(state = .x, year = year)) %>%
    st_transform(target_crs) # Project to our target CRS after all states are combined.
}

# --- Download PUMAs (Our Target Geography) ---
pumas_sf <- download_national_file(pumas, 2020)

# --- Download Source Geographies ---
# We download each granular geography and create a `GISJOIN` column that will match
# the `GISJOIN` identifier in our main `nhgis_data` file. This is the key to joining
# our spatial results back to the income data later.
tracts_sf <- download_national_file(tracts, 2020) %>%
  mutate(GISJOIN = paste0("G", GEOID))

county_subdivisions_sf <- download_national_file(county_subdivisions, 2020) %>%
  mutate(GISJOIN = paste0("G", GEOID))

school_districts_sf <- download_national_file(school_districts, 2020) %>%
  mutate(GISJOIN = paste0("G", GEOID))

# ZCTAs (ZIP Code Tabulation Areas) are a national file and don't require the state-by-state loop.
message("-> Downloading national file for: zctas")
zctas_sf <- zctas(year = 2020) %>%
  st_transform(target_crs) %>%
  mutate(GISJOIN = paste0("G", ZCTA5CE20)) # The ZCTA identifier column has a different name.

# Combine the granular shapefiles into a single list for easier processing in the next step.
source_shapefiles <- list(
  tract = tracts_sf,
  county_subdivision = county_subdivisions_sf,
  school_district = school_districts_sf,
  zcta = zctas_sf
)

message("All shapefiles downloaded and transformed to a consistent CRS.")


# ==== 3. SPATIAL INTERSECTION & PREDOMINANT PUMA CALCULATION ====
# Purpose: To determine the single "best" PUMA for each granular geography.
#
# Logic: A single census tract can physically cross the boundary of two or more PUMAs.
# To create a simple one-to-one relationship, we find all the overlapping pieces and then
# assign the PUMA that covers the largest area of the tract. This "predominant area"
# method is a standard and robust approach in geospatial analysis.

message("Calculating predominant PUMA based on spatial intersection area...")

# This function automates the intersection and predominance calculation.
calculate_predominant_puma_gis <- function(source_sf, pumas_sf) {
  # Step 1: Find all geometric overlaps between the source shapes and the PUMA shapes.
  # The result is a new set of smaller polygons representing every piece of overlap.
  message("   ...performing intersection...")
  intersections <- st_intersection(source_sf, pumas_sf)
  
  # Step 2: Calculate the area of each new, smaller overlap polygon.
  # This works accurately because we are using an equal-area CRS.
  intersections$intersection_area <- st_area(intersections)
  
  # Step 3: Find the predominant PUMA for each source geography.
  #  - `st_drop_geometry()`: Temporarily converts the spatial data to a regular table
  #    to make grouping and summarizing much faster.
  #  - `group_by(GISJOIN)`: Groups all the overlapping pieces that belong to the same
  #    original source geography (e.g., all parts of a single census tract).
  #  - `slice_max()`: Within each group, this finds the single row (the single overlap)
  #    that has the largest `intersection_area`. This is the "winning" PUMA.
  #  - `select()`: Creates a clean final table with just the source GISJOIN and the
  #    `GEOID20` of the winning PUMA.
  message("   ...finding predominant PUMA by area...")
  predominant_lookup <- intersections %>%
    st_drop_geometry() %>%
    group_by(GISJOIN) %>%
    slice_max(order_by = intersection_area, n = 1, with_ties = FALSE) %>%
    ungroup() %>%
    select(GISJOIN, derived_PUMA = GEOID20)
  
  return(predominant_lookup)
}

# Apply the function to each of our source shapefiles to generate lookup tables.
all_puma_lookups_gis <- lapply(source_shapefiles, calculate_predominant_puma_gis, pumas_sf = pumas_sf)

# Combine all the individual lookup tables into one master lookup table.
master_puma_lookup_gis <- bind_rows(all_puma_lookups_gis) %>%
  distinct(GISJOIN, .keep_all = TRUE)

message(paste("Created a master PUMA lookup table with", nrow(master_puma_lookup_gis), "unique geographic areas."))


# ==== 4. INTEGRATE THE PUMA LOOKUP WITH THE MAIN DATABASE ====
# Purpose: To merge our new, accurate PUMA assignments back into the main dataset.
#
# Logic: We use a `left_join` to bring the `derived_PUMA` from our spatial analysis
# into the main `nhgis_data` table, matching on the `GISJOIN` key. The `coalesce()`
# function is used to fill in the PUMA column: if the original `PUMA` column is `NA`,
# it uses our `derived_PUMA`; otherwise, it keeps the original value.

message("Joining the GIS-based PUMA lookup table with the main NHGIS summary data...")

nhgis_data_harmonized <- nhgis_data %>%
  mutate(PUMA = as.character(PUMA)) %>%
  left_join(master_puma_lookup_gis, by = "GISJOIN") %>%
  mutate(
    PUMA = coalesce(PUMA, derived_PUMA)
  ) %>%
  select(-derived_PUMA) # Clean up the temporary column.

message("PUMA values have been successfully integrated.")


# ==== 5. FINAL CHECK AND SAVE ====
# Purpose: To verify the results and save the final, enriched dataset.
message("Final check for NA PUMAs after integration...")
na_summary <- nhgis_data_harmonized %>%
  # We filter out geographies where a PUMA is not applicable (like states) or
  # where the PUMA was already known.
  filter(!geography_level %in% c("State", "County", "Unknown", "PUMA", "Urban Area", "CBSA")) %>%
  group_by(geography_level) %>%
  summarise(
    total_rows = n(),
    na_pumas = sum(is.na(PUMA)),
    .groups = 'drop'
  )
print("NA PUMA counts for granular geographies:")
print(na_summary)

output_dir <- here("01_data", "processed")
dir.create(output_dir, showWarnings = FALSE, recursive = TRUE)
rds_output_path <- here(output_dir, "geographic_income_database_nhgis_summary_with_puma.rds")
csv_output_path <- here(output_dir, "geographic_income_database_nhgis_summary_with_puma.csv")

message(paste("Saving updated RDS file to:", rds_output_path))
saveRDS(nhgis_data_harmonized, rds_output_path)
message("Updated RDS file saved.")

message(paste("Saving updated CSV file to:", csv_output_path))
write.csv(nhgis_data_harmonized, csv_output_path, row.names = FALSE)
message("Updated CSV file saved.")

message("Script execution complete.")

# -----------------------------------------------------------------------------
# 1. SETUP: INSTALL AND LOAD NECESSARY PACKAGES

library(dplyr)
library(readr)
library(purrr)


# -----------------------------------------------------------------------------

# -----------------------------------------------------------------------------
# To use the IPUMS API, you need to register for an account on the IPUMS
# website (https://account.ipums.org/api_keys) and get your API key.
# It is best practice to set this as an environment variable rather than
# hardcoding it in your script. You mentioned your key is already set up.

# set_ipums_api_key("YOUR_API_KEY_HERE", save = TRUE)

# -----------------------------------------------------------------------------
# 3. DEFINE THE DATA EXTRACT FROM NHGIS
# -----------------------------------------------------------------------------
# We are updating this to use the 2019-2023 5-year ACS data and the precise
# technical geographic level names required by the IPUMS API.

# Define the extract object
nhgis_extract <- define_extract_nhgis(
  description = "Median Household Income for Various Geographies 2019-2023",
  datasets = ds_spec(
    "2019_2023_ACS5a", # Using the 2019-2023 5-year ACS data
    data_tables = "B19013", # Table for Median Household Income
    # Using the exact technical geographic level names for the API request:
    geog_levels = c(
      "tract",
      "blck_grp",
      "cty_sub",
      "place",
      "sd_elm",
      "sd_sec",
      "sd_uni",
      "puma",
      "cbsa",
      "urb_area",
      "zcta",
      "county", # Included for context
      "state"   # Included for context
    )
  )
)

# -----------------------------------------------------------------------------
# 4. SUBMIT AND DOWNLOAD THE EXTRACT
# -----------------------------------------------------------------------------
# This part of the script sends the extract definition to the NHGIS servers.
# The system will prepare the data, and then we can download it.
# This process can take a few minutes, especially with many geographic levels.

# Submit the extract to the NHGIS server
submitted_extract <- submit_extract(nhgis_extract)

# Wait for the extract to be processed by the server
# This function will check the status of the extract every 30 seconds
downloadable_extract <- wait_for_extract(submitted_extract)

# Download the data to a temporary directory
# The `download_extract` function returns the path to the downloaded .zip file
data_files <- download_extract(downloadable_extract)

# -----------------------------------------------------------------------------
# 5. READ AND COMBINE THE DATA
# -----------------------------------------------------------------------------
# The downloaded .zip file contains separate CSV files for each geographic level.
# We need to read each of these files and then combine them into a single,
# tidy data frame.

# Create a temporary directory to unzip the files
data_path <- file.path(tempdir(), "nhgis_unzipped")
unzip(data_files, exdir = data_path)

# Get the list of CSV files from the unzipped directory
csv_files <- list.files(data_path, pattern = "\\.csv$", full.names = TRUE)

# Read each CSV and store it in a list of data frames
# Using `map` from the 'purrr' package, which is a tidyverse alternative to lapply
all_data <- map(csv_files, read_nhgis)

# Combine all the data frames into one
# The `bind_rows` function from dplyr is smart enough to handle columns that
# might not be present in all files.
combined_data <- bind_rows(all_data)

# -----------------------------------------------------------------------------
# 6. CLEAN AND PREPARE THE FINAL DATASET
# -----------------------------------------------------------------------------
# Now we will clean up the combined dataset to make it more user-friendly.
# This includes renaming columns, selecting the most important variables, and
# creating a clear 'geography_level' column based on the new geographies.

final_database <- combined_data %>%
  # The column with the estimate for B19013 is 'B19013_001E'
  # The column with the margin of error is 'B19013_001M'
  rename(
    median_household_income = B19013_001E,
    median_household_income_moe = B19013_001M
  ) %>%
  # Create a column to identify the geographic level of the row
  # The order is important, from most granular to least granular.
  mutate(
    geography_level = case_when(
      !is.na(BLKGRPA) ~ "Block Group",
      !is.na(TRACTA) ~ "Census Tract",
      !is.na(ZCTA) ~ "ZCTA",
      !is.na(PLACEA) ~ "Place",
      !is.na(COUSUB) ~ "County Subdivision",
      !is.na(SDELM) ~ "Elementary School District",
      !is.na(SDSEC) ~ "Secondary School District",
      !is.na(SDUNI) ~ "Unified School District",
      !is.na(PUMAA) ~ "PUMA",
      !is.na(CBSAA) ~ "CBSA",
      !is.na(UACE) ~ "Urban Area",
      !is.na(COUNTY) ~ "County",
      !is.na(STATE) ~ "State",
      TRUE ~ "Unknown"
    )
  ) %>%
  # Select the most relevant columns for our reference database
  select(
    GISJOIN,
    YEAR,
    STATE,
    COUNTY,
    geography_level,
    NAME,
    median_household_income,
    median_household_income_moe
  )

# -----------------------------------------------------------------------------
# 7. SAVE AND EXPLORE THE FINAL DATABASE
# -----------------------------------------------------------------------------
# The 'final_database' is now ready to be used. You can save it as a CSV
# or an R data file for easy access in the future.

# Save as a CSV file
write.csv(final_database, "nhgis_median_income_database_2019_2023.csv", row.names = FALSE)

# Save as an R data file (which is more efficient for R)
saveRDS(final_database, "nhgis_median_income_database_2019_2023.rds")

# You can now view the first few rows of your new database
print("--- First 6 rows of the final database ---")
print(head(final_database))

# And get a summary of the data
print("--- Summary of the final database ---")
summary(final_database)

# Example of how to pull data for a specific geography
# For example, all PUMAs in California
pumas_in_california <- final_database %>%
  filter(geography_level == "PUMA", STATE == "California")

print("--- Example: PUMAs in California ---")
print(head(pumas_in_california))

# You are now ready to merge this database with your other datasets using the

# 3. DEFINE THE DATA EXTRACT FROM NHGIS

# We are updating this to use the 2019-2023 5-year ACS data and the precise
# technical geographic level names required by the IPUMS API.

# Define the extract object
nhgis_extract <- define_extract_nhgis(
  description = "Median Household Income for Various Geographies 2019-2023",
  datasets = ds_spec(
    "2019_2023_ACS5a", # Using the 2019-2023 5-year ACS data
    data_tables = "B19013", # Table for Median Household Income
    # Using the exact technical geographic level names for the API request:
    geog_levels = c(

      "blck_grp",
      "cty_sub",

      "sd_elm",

      "sd_uni",

      "cbsa",
      "urb_area",
      "zcta",

      "state"   # Included for context
    )
  )

# -----------------------------------------------------------------------------

# -----------------------------------------------------------------------------
# This part of the script sends the extract definition to the NHGIS servers.
# The system will prepare the data, and then we can download it.

# Submit the extract to the NHGIS server
submitted_extract <- submit_extract(nhgis_extract)

# Wait for the extract to be processed by the server
# This function will check the status of the extract every 30 seconds

# Download the data to a temporary directory
# The `download_extract` function returns the path to the downloaded .zip file
data_files <- download_extract(downloadable_extract)

# -----------------------------------------------------------------------------
# 5. READ AND COMBINE THE DATA
# -----------------------------------------------------------------------------
# The downloaded .zip file contains separate CSV files for each geographic level.
# We need to read each of these files and then combine them into a single,
# tidy data frame.

# Create a temporary directory to unzip the files
data_path <- file.path(tempdir(), "nhgis_unzipped")
unzip(data_files, exdir = data_path)

# Use `recursive = TRUE` to find the CSV files even if they are in a subdirectory.
csv_files <- list.files(data_path, pattern = "\\.csv$", full.names = TRUE, recursive = TRUE)

# Add a check to ensure that files were actually found.
if (length(csv_files) == 0) {
  stop("No CSV files were found after unzipping the extract. Please check the contents of the downloaded zip file.")
}

# **FIX:** Read all columns as character type to prevent type conflicts during binding.
# The `col_types = cols(.default = "c")` argument ensures all columns are read as text.
all_data <- map(csv_files, ~ read_nhgis(., col_types = cols(.default = "c")))

# Combine all the data frames into one
# This will now work because all columns have a consistent character type.
combined_data <- bind_rows(all_data)

# -----------------------------------------------------------------------------
# 6. CLEAN AND PREPARE THE FINAL DATASET
# -----------------------------------------------------------------------------
# Now we will clean up the combined dataset to make it more user-friendly.

# Using the exact column names confirmed from the codebooks.
estimate_col <- "ASQPE001"
moe_col <- "ASQPM001"

# Check if these columns exist before proceeding.
if (!all(c(estimate_col, moe_col, "NAME_E", "NAME_M") %in% names(combined_data))) {
  stop(paste("The expected columns", estimate_col, ",", moe_col, ", NAME_E, and NAME_M were not found in the combined data. Please verify the extract contents."))
}

final_database <- combined_data %>%
  # Rename the columns using the exact names from the codebook.
  rename(
    median_household_income_chr = !!estimate_col,
    median_household_income_moe_chr = !!moe_col
  ) %>%
  # Convert income and MOE columns to numeric now that they are combined.
  # Also, create a single NAME column from NAME_E and NAME_M.
  mutate(
    median_household_income = as.numeric(median_household_income_chr),
    median_household_income_moe = as.numeric(median_household_income_moe_chr),
    NAME = coalesce(NAME_E, NAME_M) # **FIX:** Create a single NAME column
  ) %>%
  # Create a column to identify the geographic level of the row
  # The order is important, from most granular to least granular.
  mutate(
    geography_level = case_when(
      !is.na(BLKGRPA) ~ "Block Group",
      !is.na(TRACTA) ~ "Census Tract",
      !is.na(ZCTA5A) ~ "ZCTA",
      !is.na(PLACEA) ~ "Place",
      !is.na(COUSUBA) ~ "County Subdivision",
      !is.na(SDELMA) ~ "Elementary School District",
      !is.na(SDSECA) ~ "Secondary School District",
      !is.na(SDUNIA) ~ "Unified School District",
      !is.na(PUMAA) ~ "PUMA",
      !is.na(CBSAA) ~ "CBSA",
      !is.na(UAA) ~ "Urban Area",
      !is.na(COUNTYA) ~ "County",
      !is.na(STATEA) ~ "State",
      TRUE ~ "Unknown"
    )
  ) %>%
  # Select the most relevant columns for our reference database
  select(
    GISJOIN,
    YEAR,
    STATE,
    COUNTY,
    geography_level,
    NAME,
    median_household_income,
    median_household_income_moe
  )

# -----------------------------------------------------------------------------
# 7. SAVE AND EXPLORE THE FINAL DATABASE
# -----------------------------------------------------------------------------

# or an R data file for easy access in the future.

# Save as a CSV file
write.csv(final_database, "nhgis_median_income_database_2019_2023.csv", row.names = FALSE)

# Save as an R data file (which is more efficient for R)
saveRDS(final_database, "nhgis_median_income_database_2019_2023.rds")

print("--- First 6 rows of the final database ---")
print(head(final_database))

# And get a summary of the data

## Purpose: Calculates weighted median household income for each PUMA-level

## Date Created: 2025-08-18
## Last Modified: 2025-09-22 (Updated to use 7-digit PUMA_GEOID)

# ==== 1. LOAD IPUMS ACS MICRODATA ====
input_microdata_path <- here("01_data", "processed", "ipums_acs_2019_2023_microdata.rds")

if (!file.exists(input_microdata_path)) {
  stop(paste("Error: Microdata file not found at", input_microdata_path,
             ". Please ensure Task I-B has been completed successfully."))
}

message(paste("Loading IPUMS ACS microdata from:", input_microdata_path))
ipums_microdata <- readRDS(input_microdata_path)
message("Microdata loaded successfully.")

# ==== 2. DEFINE STANDARD INCOME BRACKETS (NHGIS B19001) ====
income_breaks <- c(0, 9999, 14999, 19999, 24999, 29999, 34999, 39999, 44999,
                   49999, 59999, 74999, 99999, 124999, 149999, 199999,
                   249999, Inf)

income_labels <- c(
  "Less than $10,000", "$10,000 to $14,999", "$15,000 to $19,999",
  "$20,000 to $24,999", "$25,000 to $29,999", "$30,000 to $34,999",
  "$35,000 to $39,999", "$40,000 to $44,999", "$45,000 to $49,999",
  "$50,000 to $59,999", "$60,000 to $74,999", "$75,000 to $99,999",
  "$100,000 to $124,999", "$125,000 to $149,999", "$150,000 to $199,999",
  "$200,000 to $249,999", "$250,000 or more"
)

weighted_median <- function(x, w) {
  if (length(x) == 0 || sum(w) == 0) return(NA)
  df <- data.frame(x = x, w = w) %>% arrange(x)
  df$w_cumsum <- cumsum(df$w)
  median_weight <- sum(df$w) / 2
  df$x[which(df$w_cumsum >= median_weight)[1]]
}

# ==== 3. ASSIGN BRACKETS AND CALCULATE MEDIANS ====
message("Assigning households to income brackets...")
microdata_with_brackets <- ipums_microdata %>%
  mutate(
    Income_Bracket_ID = cut(
      HHINCOME,
      breaks = income_breaks,
      labels = income_labels,
      right = FALSE,
      include.lowest = TRUE
    )
  )

message("Calculating weighted median income for each PUMA-bracket group...")
# **CORRECTED** to group by the full 7-digit PUMA_GEOID
puma_bracket_medians <- microdata_with_brackets %>%
  filter(!is.na(Income_Bracket_ID)) %>%
  group_by(PUMA_GEOID, Income_Bracket_ID) %>%
  summarise(
    Calculated_Median_Income = weighted_median(HHINCOME, HHWT),
    .groups = 'drop'
  )

# ==== 4. CREATE AND SAVE LOOKUP TABLE ====

  rename(PUMA_Bracket_Median_HHINCOME = Calculated_Median_Income)

message("RDS file saved.")

# ===== 1.3. Helper Function to Download and Prepare Geographies =====
download_and_prep_state_file <- function(tigris_function, year, states_list, geo_name, cb_flag, type_arg = NULL) {
  download_type <- if (cb_flag) "Cartographic Boundary" else "Full Resolution"
  # Create a more descriptive name if a 'type' is specified (e.g., for school districts)
  full_geo_name <- if (!is.null(type_arg)) paste(type_arg, "school districts") else geo_name
  
  message(paste("   -> Downloading", full_geo_name, "for", year, "by state (", download_type, ")..."))
  
  # Use purrr::map to loop over each state FIPS code and download the data.
  results <- purrr::map(states_list, function(st) {
    log_entry_base <- list(year = year, geography = full_geo_name, state = st, type = "State-by-State")
    
    sf_data <- tryCatch({
      # Call the appropriate tigris function, passing the 'type' argument only if it's provided.
      if (is.null(type_arg)) {
        tigris_function(state = st, year = year, cb = cb_flag, progress_bar = TRUE)
      } else {
        tigris_function(state = st, year = year, cb = cb_flag, type = type_arg, progress_bar = TRUE)
      }
    }, error = function(e) {
      # On failure, log the error and return NULL.
      download_log <<- append(download_log, list(c(log_entry_base, status = "FAILED", error_message = e$message)))
      return(NULL)
    })
    
    # On success, log the success and return the downloaded data.
    if (!is.null(sf_data)) {
      download_log <<- append(download_log, list(c(log_entry_base, status = "SUCCESS", error_message = NA)))
    }
    return(sf_data)
  })
  
  # Combine all successfully downloaded state data frames into one national file.
  national_sf <- compact(results) %>% bind_rows()
  
  if (nrow(national_sf) == 0) {
    warning(paste("No data was successfully downloaded for", full_geo_name, "in", year, "."))
    return(NULL)
  }
  
  # Transform the combined shapefile to the target CRS.
  return(st_transform(national_sf, target_crs))
}

# ===== 1.4. Helper Functions for Robust GEOID Renaming =====
# These functions find common GEOID variants and rename them to a standard `TL_GEO_ID`.
rename_geoid_10 <- function(sf_obj) {
  if (is.null(sf_obj)) return(NULL)
  if ("GEOID10" %in% names(sf_obj)) {
    rename(sf_obj, TL_GEO_ID = GEOID10)
  } else if ("ZCTA5CE10" %in% names(sf_obj)) {
    rename(sf_obj, TL_GEO_ID = ZCTA5CE10)
  } else if ("GEOID" %in% names(sf_obj)) {
    rename(sf_obj, TL_GEO_ID = GEOID)
  } else {
    warning("Could not find a standard 2010-era GEOID column.")
    sf_obj
  }
}

rename_geoid_20 <- function(sf_obj) {
  if (is.null(sf_obj)) return(NULL)
  if ("GEOID20" %in% names(sf_obj)) {
    rename(sf_obj, TL_GEO_ID = GEOID20)
  } else if ("ZCTA5CE20" %in% names(sf_obj)) {
    rename(sf_obj, TL_GEO_ID = ZCTA5CE20)
  } else if ("GEOID" %in% names(sf_obj)) {
    rename(sf_obj, TL_GEO_ID = GEOID)
  } else {
    warning("Could not find a standard 2020-era GEOID column.")
    sf_obj
  }
}

# ===== 1.2. Updated Helper Function =====
process_crosswalk_file <- function(file_info, geography_name) {
  message(paste("--- Processing crosswalk for:", geography_name, "---"))
  
  file_path <- file.path(crosswalk_base_dir, file_info$path)
  
  if (!file.exists(file_path)) {
    warning(paste("File not found for", geography_name, "at path:", file_path, ". Skipping."))
    return(NULL)
  }
  
  # Read all columns as character to preserve leading zeros in GEOIDs
  crosswalk_data <- read_delim(file_path, delim = "|", col_types = cols(.default = "c"))
  
  # Ensure the specific columns exist before proceeding
  required_cols <- c(file_info$geoid_10_col, file_info$geoid_20_col, "AREALAND_PART")
  if (!all(required_cols %in% names(crosswalk_data))) {
    warning(paste("Could not find required columns for", geography_name, ". Expected:", paste(required_cols, collapse=", "), ". Skipping this file."))
    return(NULL)
  }
  
  processed_df <- crosswalk_data %>%
    select(
      GEOID10 = all_of(file_info$geoid_10_col),
      GEOID20 = all_of(file_info$geoid_20_col),
      AREALAND_PART
    ) %>%
    mutate(
      GEOGRAPHY_TYPE = geography_name,
      AREALAND_PART = as.numeric(AREALAND_PART)
    ) %>%
    filter(!is.na(GEOID10) & !is.na(GEOID20))
  
  message(paste("... Successfully processed", nrow(processed_df), "relationships for", geography_name, "."))
  return(processed_df)
}

# ===== 2.1. Function for Geographies WITH a Crosswalk (Tracts, COUSUBs, ZCTAs) =====
harmonize_and_assign <- function(granular_sf_2010, crosswalk, pumas_sf_2020, geo_type) {
  message(paste("--- Starting harmonization (with crosswalk) for:", geo_type, "---"))
  
  granular_sf_2010_clean <- granular_sf_2010 %>%
    mutate(TL_GEO_ID = as.character(trimws(TL_GEO_ID)))
  
  crosswalk_filtered <- crosswalk %>%
    filter(GEOGRAPHY_TYPE == geo_type) %>%
    mutate(GEOID10 = as.character(trimws(GEOID10))) %>%
    select(GEOID10, GEOID20)
  
  granular_sf_with_xwalk <- granular_sf_2010_clean %>%
    left_join(crosswalk_filtered, by = c("TL_GEO_ID" = "GEOID10"))
  
  na_matches <- sum(is.na(granular_sf_with_xwalk$GEOID20))
  if (na_matches > 0) {
    message(paste("Warning:", na_matches, "of", nrow(granular_sf_with_xwalk), "records in", geo_type, "failed to find a match in the crosswalk."))
  }
  
  pumas_2020_id <- pumas_sf_2020 %>% select(PUMA_GEOID20 = TL_GEO_ID)
  
  message("... Performing spatial intersection...")
  intersection_sf <- st_intersection(st_make_valid(granular_sf_with_xwalk), st_make_valid(pumas_2020_id))
  
  intersection_sf$intersection_area <- st_area(intersection_sf)
  
  lookup_table <- st_drop_geometry(intersection_sf) %>%
    group_by(TL_GEO_ID) %>%
    slice_max(order_by = intersection_area, n = 1, with_ties = FALSE) %>%
    ungroup() %>%
    select(GEOID10 = TL_GEO_ID, GEOID20, PUMA_GEOID20) %>%
    mutate(GEOGRAPHY_TYPE = geo_type)
  
  message(paste("--- Completed assignment for:", geo_type, "---"))
  return(lookup_table)
}

# ===== 2.2. Function for Geographies WITHOUT a Crosswalk (School Districts) =====
assign_puma_spatially_only <- function(granular_sf_2010, pumas_sf_2020, geo_type) {
  message(paste("--- Starting SPATIAL-ONLY assignment for:", geo_type, "---"))
  
  pumas_2020_id <- pumas_sf_2020 %>% select(PUMA_GEOID20 = TL_GEO_ID)
  
  message("... Performing direct spatial intersection...")
  intersection_sf <- st_intersection(st_make_valid(granular_sf_2010), st_make_valid(pumas_2020_id))
  
  intersection_sf$intersection_area <- st_area(intersection_sf)
  
  lookup_table <- st_drop_geometry(intersection_sf) %>%
    group_by(TL_GEO_ID) %>%
    slice_max(order_by = intersection_area, n = 1, with_ties = FALSE) %>%
    ungroup() %>%
    select(GEOID10 = TL_GEO_ID, PUMA_GEOID20) %>%
    # Since there's no crosswalk, we'll assume GEOID20 is the same as GEOID10 for consistency.
    # This is a safe assumption for school districts which are relatively stable.
    mutate(GEOID20 = GEOID10, GEOGRAPHY_TYPE = geo_type)
  
  message(paste("--- Completed assignment for:", geo_type, "---"))
  return(lookup_table)
}

  select(TL_GEO_ID, !!total_hh_col, all_of(b19001_bracket_cols)) %>%
  pivot_longer(cols = all_of(b19001_bracket_cols), names_to = "bracket_col", values_to = "count") %>%
  mutate(Income_Bracket_ID = bracket_map[bracket_col], count = as.numeric(count)) %>%
  filter(!is.na(count) & count > 0) %>%
  mutate(Income_Bracket_ID = factor(Income_Bracket_ID, levels = income_brackets_16_nhgis)) %>%
  arrange(TL_GEO_ID, Income_Bracket_ID) %>%
  group_by(TL_GEO_ID) %>%
  mutate(cumulative_pct = cumsum(count) / first(!!sym(total_hh_col))) %>%
  filter(cumulative_pct >= 0.5) %>%
  slice_min(order_by = cumulative_pct, n = 1, with_ties = FALSE) %>%
  ungroup() %>%
  select(TL_GEO_ID, Median_Income_Bracket_Calculated = Income_Bracket_ID)

# ------ 3.3.1. Helper function to calculate borders within a group ------
calculate_within_borders <- function(tercile_design, income_sym, tercile_label) {
  if (is.null(tercile_design) || !inherits(tercile_design, "survey.design") || nrow(tercile_design$variables) == 0) {
    warning(paste("Invalid/empty design for", tercile_label, ". Skipping."))
    return(NULL)
  }
  
  quantile_defs <- list(
    "Groups_3" = list(probs = c(1/3, 2/3)),
    "Groups_4" = list(probs = c(1/4, 2/4, 3/4)),
    "Groups_5" = list(probs = c(1/5, 2/5, 3/5, 4/5)),
    "Groups_10" = list(probs = seq(0.1, 0.9, by = 0.1)),
    "Groups_20" = list(probs = seq(0.05, 0.95, by = 0.05))
  )
  
  print(paste("--- Calculating borders for main tercile:", tercile_label, "---"))
  
  purrr::map_dfr(quantile_defs, .id = "QuantileGroup", function(q_info) {
    q_values_raw <- tryCatch({
      survey::svyquantile(
        x = stats::as.formula(paste0("~", rlang::as_string(income_sym))),
        design = tercile_design,
        quantiles = q_info$probs,
        na.rm = TRUE,
        ci = FALSE
      )
    }, error = function(e) { NULL })
    
    if (!is.null(q_values_raw)) {
      tibble::tibble(
        QuantileProbability = q_info$probs,
        CutoffValue = q_values_raw[[1]][1,]
      )
    } else {
      tibble()
    }
  })
}

USER_HHID_VAR          <- "SERIAL"

# --- 2.2. Load Raw Data ---
raw_asec_data <- readRDS(file.path(RAW_DATA_DIR, "raw_asec_data.rds"))
raw_basic_data <- readRDS(file.path(RAW_DATA_DIR, "raw_basic_monthly_data.rds"))

# --- 2.3. USER ACTION: Clean Raw ASEC and Basic Monthly Data ---
message("Cleaning raw data...")
cleaned_asec_data <- raw_asec_data %>%
  mutate(
    HHINCOME = if_else(HHINCOME == 99999999, NA_real_, as.numeric(HHINCOME)),
    EMPSTAT = if_else(EMPSTAT == 0, NA_integer_, as.integer(EMPSTAT)),
    WHYPTLWK = if_else(WHYPTLWK == 0, NA_integer_, as.integer(WHYPTLWK)),
    WKSTAT = if_else(WKSTAT == 0, NA_integer_, as.integer(WKSTAT))
  )

cleaned_basic_data <- raw_basic_data %>%
  select(all_of(c(USER_LINKING_KEY, "UH_DSCWK_B2"))) %>%
  mutate(
    UH_DSCWK_B2 = if_else(UH_DSCWK_B2 == 0, NA_integer_, as.integer(UH_DSCWK_B2))
  )

# --- 2.4. Merge Data ---
message("Merging ASEC and Basic Monthly data...")
if(class(cleaned_asec_data[[USER_LINKING_KEY]]) != class(cleaned_basic_data[[USER_LINKING_KEY]])) {
  warning("Linking key types differ. Attempting conversion to character.")
  cleaned_asec_data[[USER_LINKING_KEY]] <- as.character(cleaned_asec_data[[USER_LINKING_KEY]])
  cleaned_basic_data[[USER_LINKING_KEY]] <- as.character(cleaned_basic_data[[USER_LINKING_KEY]])
}
merged_data <- left_join(cleaned_asec_data, cleaned_basic_data, by = USER_LINKING_KEY)

  mutate(

    is_unemployed = if_else(EMPSTAT %in% c(20, 21, 22), 1, 0, missing = NA_real_),
    is_in_labor_force = if_else(EMPSTAT %in% c(1, 10, 12, 20, 21, 22), 1, 0, missing = NA_real_),
    is_part_time_economic = if_else(WKSTAT %in% c(11:15) & WHYPTLWK %in% c(1:4), 1, 0, missing = 0),
    is_marginally_attached = if_else(UH_DSCWK_B2 %in% 1:2, 1, 0, missing = 0)
  ) %>%
  # This part calculates two potential indicators. The user selects one in the `visualize` script.
  mutate(
    unemployment_rate_u3 = if_else(AGE >= 25, is_unemployed / is_in_labor_force, NA_real_),
    unemployment_rate_u6 = if_else(AGE >= 25,
                                   (is_unemployed + is_part_time_economic + is_marginally_attached) / (is_in_labor_force + is_marginally_attached),
                                   NA_real_)
  )

# --- 2.6. Finalize and Save Processed Data ---

  USER_HHID_VAR, USER_HH_WEIGHT_VAR, USER_PERSON_WEIGHT_VAR, USER_INCOME_VAR, USER_AGE_VAR,

  "unemployment_rate_u3", "unemployment_rate_u6" # Keep the final derived variables

saveRDS(final_prepared_data, file = PROCESSED_DATA_FILE)

get_ui_data <- function(level, source, topic, subtopic = NULL, filters = list(year = USER_YEAR), ...) {
  message(sprintf("-> Downloading data for: %s (source: %s, level: %s)", topic, source, level))
  
  res <- tryCatch({
    httr::with_config(httr::timeout(300), {
      get_education_data(level = level,
                         source = source,
                         topic = topic,
                         subtopic = subtopic,
                         filters = filters
      )
    })
  }, error = function(e) e)
  
  if (inherits(res, "error")) {
    message(paste("\n--- API DOWNLOAD FAILED for topic:", topic, "---\nError:", res$message))
    return(NULL)
  }
  
  message(paste("   ...Success. Downloaded", nrow(res), "rows."))
  return(res)
}

# Helper to correctly aggregate disaggregated data to the school level
collapse_school_totals <- function(df) {
  if (!"ncessch" %in% names(df)) return(df)
  group_keys <- intersect(c("ncessch", "year"), names(df))
  analysis_cols <- c("enrollment", "days_suspended", "students_chronically_absent")
  
  cols_to_sum <- intersect(analysis_cols, names(df))
  # All other columns, including identifiers, will be carried forward
  cols_to_first <- setdiff(names(df), c(group_keys, cols_to_sum))
  
  df %>%
    group_by(across(all_of(group_keys))) %>%
    summarise(
      across(all_of(cols_to_sum), ~ sum(as.numeric(.x), na.rm = TRUE)),
      across(all_of(cols_to_first), ~ first(na.omit(.x))), 
      .groups = "drop"
    )
}

## Script: UIED_gfc_acquire.R
## Purpose: Downloads specified datasets for Teacher Certification, Internet,
##          and Advanced Coursework from the Urban Institute Education Data portal.
## Author: Max Goshert, EPAG / Gemini
## Date Created: 2025-10-02
## Last Modified: 2025-10-02 (Shortened indicator name to prevent long path errors)

# ==== 0. SETUP ====
rm(list = ls()); gc()
library(educationdata); library(dplyr); library(here); library(httr); library(purrr)

# ================================================================= #
# ==== 1. SCRIPT CONFIGURATION (Generated from User Inputs) ====
# ================================================================= #

USER_YEAR <- 2017
## MODIFICATION: Shortened name to prevent Windows path length errors.
USER_INDICATOR_NAME <- "gfc" 

datasets_to_download <- list(
  directory = list(level = "school-districts", source = "ccd", topic = "directory"),
  nhgis_geo = list(level = "schools", source = "nhgis", topic = "census-2010"),
  certified_teachers = list(level = "schools", source = "crdc", topic = "teachers-staff", subtopic = NULL),
  internet_access = list(level = "schools", source = "crdc", topic = "internet-access", subtopic = NULL, filters = list(year = 2020)),
  advanced_coursework = list(level = "schools", source = "crdc", topic = "ap-ib-enrollment", subtopic = c("race", "sex")),
  enrollment_crdc = list(level = "schools", source = "crdc", topic = "enrollment", subtopic = c("race", "sex"))
)

# ================================================================= #
# ==== 2. GENERIC LOGIC (No changes needed below this line) ====
# ================================================================= #

OUTPUT_RAW_DIR <- here::here("01_data", "raw", "Urban Institute Education Data", USER_INDICATOR_NAME, USER_YEAR)
dir.create(OUTPUT_RAW_DIR, showWarnings = FALSE, recursive = TRUE)

get_ui_data <- function(level, source, topic, subtopic = NULL, filters = list(year = USER_YEAR), ...) {
  year_to_download <- filters$year %||% USER_YEAR
  message(sprintf("-> Downloading data for: %s (source: %s, level: %s, year: %d)", topic, source, level, year_to_download))
  
  res <- tryCatch({
    httr::with_config(httr::timeout(300), {
      get_education_data(level = level, source = source, topic = topic, subtopic = subtopic, filters = filters)
    })
  }, error = function(e) e)
  
  if (inherits(res, "error")) { message(paste("\n--- DOWNLOAD FAILED for topic:", topic, "---\nError:", res$message)); return(NULL) }
  message(paste("   ...Success. Downloaded", nrow(res), "rows.")); return(res)
}

raw_datasets <- purrr::map(datasets_to_download, ~do.call(get_ui_data, .x))

purrr::iwalk(raw_datasets, ~{
  if (!is.null(.x) && is.data.frame(.x)) {
    file_year <- datasets_to_download[[.y]]$filters$year %||% USER_YEAR
    file_name <- paste0("raw_uied_", .y, "_", file_year, ".rds")
    saveRDS(.x, file = file.path(OUTPUT_RAW_DIR, file_name))
    message(paste("   ...Saved:", file_name))
  }
})

message("\nData acquisition script complete.")


==== FILE: IV- Visuals/UIED/Gifted_Fiber_Certified/UIED_gfc_prepare.R ====

## WD location: 02_Scripts/IV- Variable Visual Scripts/Urban Institute Education Data/Gifted_Fiber_Certified
## Script: UIED_gfc_prepare.R
## Purpose: Prepares raw UIED data by merging, aggregating to the district level,
##          calculating specified rates, and joining with the income database.
## Author: Max Goshert, EPAG / Gemini
## Date Created: 2025-10-02
## Last Modified: 2025-10-02 (v4 - Final logic for weighted fiber rate)

# ==== 0. SETUP ====
rm(list = ls()); gc()
library(dplyr); library(readr); library(here); library(stringr); library(purrr); library(rlang)

# ================================================================= #
# ==== 1. SCRIPT CONFIGURATION ====
# ================================================================= #

USER_YEAR <- 2017
USER_INDICATOR_NAME <- "gfc"

DISAGGREGATED_DATASETS <- c("advanced_coursework", "enrollment_crdc")
VARIABLES_TO_SUM <- c(
  "enrl_IB", "enrl_gifted_talented", "enrl_AP", 
  "teachers_certified_fte", "teachers_fte_crdc", "enrollment_crdc"
)
SPECIAL_VALUES_TO_NA <- c(-1, -2, -3, -5, -6, -9, -99)

# ================================================================= #
# ==== 2. GENERIC LOGIC (with custom calculations) ====
# ================================================================= #

RAW_DATA_DIR <- here::here("01_data", "raw", "Urban Institute Education Data", USER_INDICATOR_NAME, USER_YEAR)
OUTPUT_DIR <- here::here("01_data", "processed", "Finalized RDS outputs", "Urban Institute Education Data")
dir.create(OUTPUT_DIR, showWarnings = FALSE, recursive = TRUE)
PROCESSED_DATA_FILE <- file.path(OUTPUT_DIR, paste0("uied_district_data_with_income_", USER_INDICATOR_NAME, "_", USER_YEAR, ".rds"))

raw_file_names <- c("directory", "nhgis_geo", "certified_teachers", "internet_access", "advanced_coursework", "enrollment_crdc")

raw_datasets <- purrr::map(raw_file_names, ~{
  file_year <- if (.x == "internet_access") 2020 else USER_YEAR
  readRDS(file.path(RAW_DATA_DIR, paste0("raw_uied_", .x, "_", file_year, ".rds")))
}) %>% setNames(raw_file_names)

collapse_school_totals <- function(df, vars_to_sum) {
  if (!"ncessch" %in% names(df)) return(df)
  group_keys <- intersect(c("ncessch", "year"), names(df))
  cols_to_sum <- intersect(names(df), vars_to_sum)
  cols_to_first <- setdiff(names(df), c(group_keys, cols_to_sum))
  df %>%
    group_by(across(all_of(group_keys))) %>%
    summarise(across(all_of(cols_to_sum), ~ sum(as.numeric(.x), na.rm = TRUE)),
              across(all_of(cols_to_first), ~ first(na.omit(.x))), .groups = "drop")
}

for (dataset_name in DISAGGREGATED_DATASETS) {
  raw_datasets[[dataset_name]] <- collapse_school_totals(raw_datasets[[dataset_name]], VARIABLES_TO_SUM)
}

# Isolate the 2020 internet data, keeping only school ID and the fiber variable
internet_data_2020 <- raw_datasets$internet_access %>%
  select(ncessch, sch_internet_fiber)

# Get the list of all other datasets (which are from 2017)
school_data_list_2017 <- raw_datasets[c("certified_teachers", "advanced_coursework", "enrollment_crdc", "nhgis_geo")]

# Merge all the 2017 datasets together first by school ID and year
merged_school_data_2017 <- school_data_list_2017 %>%
  reduce(full_join, by = c("ncessch", "year"))

# Now, join the 2020 internet data onto the main 2017 data using ONLY the school ID
merged_school_data <- left_join(merged_school_data_2017, internet_data_2020, by = "ncessch") %>%
  mutate(leaid = coalesce(!!!syms(grep("^leaid", names(.), value = TRUE))),
         gleaid = coalesce(!!!syms(grep("^gleaid", names(.), value = TRUE))),
         fips = coalesce(!!!syms(grep("^fips", names(.), value = TRUE))))

school_with_district_info <- merged_school_data %>% left_join(raw_datasets$directory, by = c("leaid", "year"), suffix = c("_school", ""))
schools_in_geo_districts <- school_with_district_info %>% filter(agency_type %in% c(1, 2))

district_level_data <- schools_in_geo_districts %>%
  mutate(
    across(all_of(VARIABLES_TO_SUM), ~ if_else(. %in% SPECIAL_VALUES_TO_NA, NA_real_, as.numeric(.))),
    across(c(enrl_IB, enrl_gifted_talented, enrl_AP), ~ if_else(. < 0, NA_real_, .)),
    teachers_certified_fte = if_else(!is.na(teachers_certified_fte) & !is.na(teachers_fte_crdc) & teachers_certified_fte > teachers_fte_crdc, teachers_fte_crdc, teachers_certified_fte),
    has_fiber = if_else(as.numeric(sch_internet_fiber) == 1, 1, 0, missing = 0)
  ) %>%
  filter(!is.na(enrollment_crdc) & enrollment_crdc > 0) %>%
  group_by(leaid) %>%
  summarise(
    across(all_of(VARIABLES_TO_SUM), ~ sum(.x, na.rm = TRUE), .names = "total_{.col}"),
    total_schools_with_fiber = sum(has_fiber, na.rm = TRUE),
    total_schools = n(),
    gleaid = first(na.omit(gleaid)), fips = first(na.omit(fips)), agency_level = first(na.omit(agency_level)), .groups = "drop"
  )

districts_prepped_for_join <- district_level_data %>%
  filter(!is.na(gleaid) & !is.na(fips)) %>%
  mutate(full_gleaid = paste0(str_pad(fips, 2, "left", "0"), gleaid),
         GEOGRAPHY_TYPE = case_when(agency_level == 1 ~ "Elementary School District", agency_level %in% c(2, 3, 7) ~ "Secondary School District",
                                    agency_level == 4 ~ "Unified School District", TRUE ~ NA_character_)) %>%
  filter(!is.na(GEOGRAPHY_TYPE))

geo_income_db <- readRDS(here::here("01_data", "processed", "geographic_income_database_harmonized.rds"))
geo_income_db_prepped <- geo_income_db %>%
  filter(geography_level %in% c("Elementary School District", "Secondary School District", "Unified School District")) %>%
  select(TL_GEO_ID, GEOGRAPHY_TYPE = geography_level, median_household_income_final)

data_with_income <- districts_prepped_for_join %>% left_join(geo_income_db_prepped, by = c("full_gleaid" = "TL_GEO_ID", "GEOGRAPHY_TYPE")) %>%
  filter(!is.na(median_household_income_final))

final_analysis_data <- data_with_income %>%
  mutate(
    ## MODIFICATION: Calculate fiber_internet_rate here at the district level.
    certified_teacher_rate = total_teachers_certified_fte / total_teachers_fte_crdc,
    fiber_internet_rate = total_schools_with_fiber / total_schools,
    rate_ib = if_else(total_enrollment_crdc > 0, total_enrl_IB / total_enrollment_crdc, NA_real_),
    rate_gt = if_else(total_enrollment_crdc > 0, total_enrl_gifted_talented / total_enrollment_crdc, NA_real_),
    rate_ap = if_else(total_enrollment_crdc > 0, total_enrl_AP / total_enrollment_crdc, NA_real_)
  ) %>%
  rowwise() %>%
  mutate(
    rates_vector = list(c(rate_ib, rate_gt, rate_ap)),
    positive_rates = list(rates_vector[rates_vector > 0 & !is.na(rates_vector)]),
    advanced_coursework_rate = if(length(positive_rates) > 0) mean(positive_rates) else 0
  ) %>%
  ungroup() %>%
  select(-rates_vector, -positive_rates) %>%
  mutate(across(c(certified_teacher_rate, advanced_coursework_rate, fiber_internet_rate), ~if_else(is.nan(.) | is.infinite(.), NA_real_, .)))

saveRDS(final_analysis_data, file = PROCESSED_DATA_FILE)
message(paste("Data preparation complete. Final file saved to:", PROCESSED_DATA_FILE))
